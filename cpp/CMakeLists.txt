cmake_minimum_required(VERSION 3.20)
project(flow_cpp VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(PYBIND11_FINDPYTHON ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/source_hash.py"
  OUTPUT_VARIABLE FLOW_SOURCE_HASH OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY)
set(FLOW_EFFECTIVE_FLAGS "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE} -ffp-contract=off")
configure_file(src/build_info.h.in generated/build_info.h @ONLY)
pybind11_add_module(flow_cpp NO_EXTRAS src/bindings.cpp src/diffusion.cpp)
target_include_directories(flow_cpp PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(flow_cpp PRIVATE -ffp-contract=off)
endif()
install(TARGETS flow_cpp LIBRARY DESTINATION .)
